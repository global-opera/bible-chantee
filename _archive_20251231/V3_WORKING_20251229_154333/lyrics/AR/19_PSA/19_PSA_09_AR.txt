[TITLE]
# يا رب، أنت ملجأي

[LYRICS]
# يا رب، أنت ملجأي

[المقطع الأول]  
أحمدُكَ بكل قلبي،  
أحدِّثُ عن عجائبك،  
فرحٌ في روحي ينمو،  
لأنك لي يا عليَّ،  
تُهزم أعدائي،  
تسقط قواهم،  
تجلس على كرسي الحق،  
تُدين الشعوب بالعدل.

[اللازمة]  
يا رب، أنت ملجأي،  
في أوقات الضيق،  
تسمع صراخي،  
ترفعني من الموت.  
أرنمُ لك يا إلهي،  
في صهيون،  
أخبرُ الأمم بأفعالك،  
أنت لي الخلاص.

[المقطع الثاني]  
تذكرُ المساكين،  
لا تنسى آلامهم،  
ترفعُني من الهاوية،  
تريحُني في ظلامي.  
يا رب، أنت القاضي،  
تُهلك الأشرار،  
تُثبتُ الحق في الأرض،  
وتجعل المساكين فرحين.

[اللازمة]  
يا رب، أنت ملجأي،  
في أوقات الضيق،  
تسمع صراخي،  
ترفعني من الموت.  
أرنمُ لك يا إلهي،  
في صهيون،  
أخبرُ الأمم بأفعالك،  
أنت لي الخلاص.

[الجسر]  
قم يا رب، لا يعتزّ الإنسان،  
اجعل على الأمم رعباً،  
ليعلموا أنهم بشر،  
لأنك لا تنسى الضعيف،  
ترجع الأشرار إلى الهاوية،  
ورجاء البائسين لا يخيب.

[اللازمة]  
يا رب، أنت ملجأي،  
في أوقات الضيق،  
تسمع صراخي،  
ترفعني من الموت.  
أرنمُ لك يا إلهي،  
في صهيون،  
أخبرُ الأمم بأفعالك،  
أنت لي الخلاص.
