[TITLE] مذبح العبادة

[LYRICS]
[TITLE] مذبح العبادة

[LYRICS]

الآية 1:  
يا ربّ، قد صنعتَ مذبحَك،  
من خشبِ السنطِ، طولهُ خمسُ أذرعٍ،  
مربّعٌ أمامَ عرشِك،  
ثلاثُ أذرعٍ، في عظمةِ جلالِك.

الآية 2:  
قرونٌ على زوايا المذبح،  
مغشّاةٌ بنحاسٍ، أشرفُ من كلّ ما في الوجود،  
قدورٌ لجمعِ الرماد،  
وأدواتٌ بيدِ خدمتِك، يا من تهبُ النور.

الكورس:  
مذبحُ العبادة، في حضرةِ القداسة،  
نرفعُ إليكَ قلوبَنا،  
ونقدّمُ ذبيحتنا،  
يا مَن أتيتَ بالنورِ،  
أنتَ الغافرُ، أنتَ المعبودُ.

الآية 3:  
شبكةٌ من النحاسِ، في منتصفِ المذبح،  
أربعةُ حلقاتٍ، تحملُ كلَّ التضحيات،  
عصوان من خشبِ السنط،  
مغشّاةٌ بنحاسٍ، تُحملُ بيدَيْ خدامِك.

الآية 4:  
الدّارُ مقدّسةٌ، بأستارٍ مبرومة،  
مئة ذراعٍ طولًا، وحينَ تشرقُ الشمس،  
أعمدةٌ من نحاسٍ، وقواعدُ من فضة،  
نرفعُ أصواتَنا، في ترنيمِ الشكر.

الكورس:  
مذبحُ العبادة، في حضرةِ القداسة،  
نرفعُ إليكَ قلوبَنا،  
ونقدّمُ ذبيحتنا،  
يا مَن أتيتَ بالنورِ،  
أنتَ الغافرُ، أنتَ المعبودُ.

الجسر:  
يا إلهَ العهودِ، يا مَن نُخاطبُكَ،  
تحتَ الحجابِ نلتقي، في خيمةِ الاجتماع،  
نضيءُ السرجَ، زيتُ الزيتونِ نقياً،  
نُقدّمُ لكَ كلَّ ما في قلوبِنا،  
يا مَن تسكنُ في أعماقِ الأرواحِ.

الكورس:  
مذبحُ العبادة، في حضرةِ القداسة،  
نرفعُ إليكَ قلوبَنا،  
ونقدّمُ ذبيحتنا،  
يا مَن أتيتَ بالنورِ،  
أنتَ الغافرُ، أنتَ المعبودُ.

الختام:  
في كلِّ أجيالِنا، نُقيمُ العهدَ،  
يا ربَّ المجدِ، نُسبّحُكَ،  
لتكن ذبيحتنا، في ذاكَ اليومِ،  
مذبحُ العبادة، حتى النهاية.
