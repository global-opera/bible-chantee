// ================== ÉTAT DU PLAYER ================== if (typeof currentBook === 'undefined') { var currentBook = "01"; // valeur par défaut, sera écrasée par ton openBook() } if (typeof currentChapter === 'undefined') { var currentChapter = 1; } if (typeof lyricsVisible === 'undefined') { var lyricsVisible = false; } // ================== PAROLES : ACCÈS AUX DONNÉES ================== function getChapterLyrics(bookNumOrKey, chapter) { if (typeof chapterLyrics === 'undefined') return null; // clé livre : "01", "02", "19", etc. const bookKey = String(bookNumOrKey).padStart(2, '0'); const bookData = chapterLyrics[bookKey] || chapterLyrics[bookNumOrKey]; if (!bookData) return null; // différentes formes possibles pour la clé de chapitre const candidates = [ chapter, String(chapter), String(chapter).padStart(2, '0'), 'Chapitre ' + chapter ]; for (const key of candidates) { if (bookData[key]) return bookData[key]; } return null; } function extractChapterTitleFromLyrics(lyrics) { if (!lyrics) return null; const lines = lyrics .split('\n') .map(l => l.trim()) .filter(l => l.length > 0); if (lines.length === 0) return null; let line = lines[0]; // Cas [TITLE] / [LYRICS] : on prend la ligne utile if (/^\[title\]/i.test(line) && lines.length > 1) { line = lines[1]; } // Cherche la partie après "–" ou "-" const m = line.match(/[–-]\s*(.+)$/); if (m && m[1].trim()) { return m[1].trim(); } // Sinon, on prend la ligne entière comme titre return line; } // ================== LECTURE D’UN CHAPITRE ================== function playChapter(chapter) { currentChapter = chapter; // fileNames et books doivent déjà exister dans ton script const fileName = fileNames[currentBook]; const padLength = (currentBook === "19") ? 3 : 2; // 3 chiffres pour les Psaumes const chapterNum = String(chapter).padStart(padLength, '0'); const mp3FileName = currentBook + "_" + fileName + "_" + chapterNum + ".mp3"; const ARCHIVE_BASE = "https://archive.org/download/bible-chantee"; const url = ARCHIVE_BASE + "/" + mp3FileName; const audio = document.getElementById('audio'); audio.src = url; audio.play().catch(err => console.error("Erreur lecture:", url, err)); const book = books.find(b => b.num === currentBook); // Titre de base : Genese - Ch.01 const chNum = chapter.toString().padStart(2, '0'); let titleText = book.name + ' - Ch.' + chNum; // On récupère les paroles pour extraire le sous-titre (titre de chanson) const lyrics = getChapterLyrics(book.num, chapter); const chapterTitle = extractChapterTitleFromLyrics(lyrics); if (chapterTitle) { titleText += ' - ' + chapterTitle; } document.getElementById('player-title').textContent = titleText; document.getElementById('chapter-select').value = chapter; document.getElementById('prev-btn').disabled = chapter <= 1; document.getElementById('next-btn').disabled = chapter >= book.chapters; // Mettre à jour les paroles dans le panneau "Paroles" updateLyrics(); } // ================== AFFICHAGE DES PAROLES ================== function updateLyrics() { const lyricsText = document.getElementById('lyricsText'); const lyricsContainer = document.getElementById('lyricsContainer'); const lyrics = getChapterLyrics(currentBook, currentChapter); if (lyrics) { // Colorer [Couplet], [Refrain], etc. let formatted = lyrics.replace(/\[([^\]]+)\]/g, '[$1]'); lyricsText.innerHTML = formatted; } else { lyricsText.innerHTML = '
