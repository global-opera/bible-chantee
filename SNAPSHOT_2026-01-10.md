## 🔧 Technical Implementation

### URL Encoding
```javascript
// Proper encoding for spaces and accents
var audioUrl = AUDIO_BASE + encodeURIComponent(prayer.audio);
var lyricsUrl = LYRICS_BASE + encodeURIComponent(prayer.lyrics);

// Example:
// "FR_Notre Pere.mp3" → "FR_Notre%20Pere.mp3"
// "FR_D'Abord Je Te Cherche.md" → "FR_D'Abord%20Je%20Te%20Cherche.md"
```

### Markdown Cleanup
```javascript
// Remove markdown markers for display
text = text.replace(/\[(VERSE|CHORUS|BRIDGE|PRE-CHORUS|INTRO|OUTRO|REFRAIN|COUPLET|PONT)\]/gi, '');
text = text.replace(/\*\*([^*]+)\*\*/g, '$1');
```

### Error Handling
```javascript
// Audio error listener
document.getElementById("audio").addEventListener("error", function(e) {
    console.error("Audio error:", e);
    console.error("Audio src:", this.src);
    console.error("Error details:", this.error);
});

// Fetch error handling
fetch(lyricsUrl)
    .then(function(r) {
        if (!r.ok) throw new Error("HTTP " + r.status);
        return r.text();
    })
    .catch(function(err) {
        console.error("Lyrics error:", err);
        lyricsBox.textContent = ui.unavailable;
    });
```
